cmake_minimum_required(VERSION 3.16)
project(SchoolBridgeLegacyShell VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Windows 8.0 API baseline. This is not runtime compatibility proof.
if(WIN32)
  add_compile_definitions(_WIN32_WINNT=0x0602 WINVER=0x0602)
endif()

find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
add_executable(SchoolBridgeLegacyShell WIN32 src/main.cpp)
target_link_libraries(SchoolBridgeLegacyShell PRIVATE ${wxWidgets_LIBRARIES})
if(MSVC)
  target_compile_options(SchoolBridgeLegacyShell PRIVATE /W4 /permissive-)
else()
  target_compile_options(SchoolBridgeLegacyShell PRIVATE -Wall -Wextra -Wpedantic)
endif()
